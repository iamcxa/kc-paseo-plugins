import fs from "node:fs";
import {DaemonClient} from "../@getpaseo/client/dist/daemon-client.js";
import {createPaseoApi} from "../@getpaseo/client/dist/index.js";
const output="/Users/kent/Project/kc-paseo-plugins/docs/dev2-poc/.spacedock-state/d7-list-chief-of-staff/evidence/cycle-2/";const initial=JSON.parse(fs.readFileSync(output+"real-list-run.json","utf8"));
const client=new DaemonClient({clientId:"d7-reconcile",url:"ws://127.0.0.1:55519/ws",reconnect:{enabled:false}});await client.connect();const api=createPaseoApi(client);
try{const targets=[];for(const id of initial.targets){const h=api.agents.ref(id);await h.refresh();if(h.status==="running"||h.status==="initializing")await h.waitForFinish(120000);await h.refresh();const page=await h.timeline.refetch({projection:"projected",limit:120});fs.writeFileSync(output+"delivered-"+id+"-timeline.json",JSON.stringify(page,null,2),{flag:"wx"});targets.push({id,status:h.status,lastUsage:h.lastUsage,entries:page.entries.length,messages:page.entries.filter(e=>(e.item as any).agentMessage?.event==="message").map(e=>({turnId:e.turnId,timestamp:e.timestamp,text:((e.item as any).agentMessage?.text??(e.item as any).text)})),assistant:page.entries.filter(e=>e.item.type==="assistant_message").map(e=>((e.item as any).agentMessage?.text??(e.item as any).text))});}
 const chief=api.agents.ref(initial.chief.agentId);await chief.refresh();const page=await chief.timeline.refetch({projection:"projected",limit:120});fs.writeFileSync(output+"final-chief-timeline.json",JSON.stringify(page,null,2),{flag:"wx"});const result={observedAt:new Date().toISOString(),targets,chief:{status:chief.status,lastUsage:chief.lastUsage,lastError:chief.lastError}};fs.writeFileSync(output+"delivery-reconciliation.json",JSON.stringify(result,null,2),{flag:"wx"});console.log(JSON.stringify({chief:result.chief,targets:targets.map(t=>({id:t.id,status:t.status,entries:t.entries,messages:t.messages.length}))}));
}finally{await api.dispose();await client.close();}
