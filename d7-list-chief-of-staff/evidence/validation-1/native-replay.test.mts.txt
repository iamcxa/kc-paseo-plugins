import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import {test} from 'node:test';
import {ChiefObserver} from './server/chief';
import {readCommanderConversation} from './shared/commander-replies';
import {COORDINATOR_VERSION} from './shared/coordinator';
const evidence=process.env.D7_EVIDENCE!;
const load=(name:string)=>JSON.parse(fs.readFileSync(path.join(evidence,'cycle-3',name),'utf8'));
const initial=load('real-list-run.json'),id=initial.chief.agentId;
const source={events:initial.events,total:'4',complete:true,rows:[]};
for(const [name,expected] of [['restart-after-list-timeline.json','calling'],['restart-after-preview-timeline.json','quiet'],['final-chief-timeline.json','quiet']])test(`candidate replays saved ${name} as ${expected} with no model call`,async()=>{
 const page=load(name);let calls=0;
 const api:any={agents:{list:async()=>({entries:[{agent:{id,cwd:'/fictional',status:'idle',labels:{role:'comms-officer','coordinator-version':COORDINATOR_VERSION}}}]}),ref:()=>({timeline:{refetch:async()=>page},run:async()=>{calls++;throw Error('No model permitted in replay');}})}};
 const observer=new ChiefObserver(async()=>source);
 const state=await observer.read(api);
 assert.equal(state.phase,expected);assert.equal(state.error,null);assert.equal(calls,0);
 const fold=await readCommanderConversation(async()=>page,id,source);
 assert.equal(fold.failed.size,0);assert.equal(fold.uncertain.size,0);
 assert.equal(fold.unresolved.length,expected==='calling'?1:0);
});
test('known 1642 item 6: resolving first list item rejects publication while three still wait',async()=>{
 const page=load('restart-after-list-timeline.json');
 const fold=await readCommanderConversation(async()=>page,id,source);
 const result=fold.unresolved[0];
 assert.equal(result.focus.targetKey,initial.events[0].key);
 const current={...source,events:initial.events.slice(1),total:'3'};
 const observer=new ChiefObserver(async()=>current);
 await assert.rejects(observer.publish({} as any,id,result.batch,result.focus),/Focus no longer awaits the Captain/);
 assert.equal(current.events.length,3);
});
test('known 1647 item 8: title-only gate version change loses an unanswered waiting gate during fold',async()=>{
 const page=load('restart-after-list-timeline.json');
 const before=await readCommanderConversation(async()=>page,id,source);
 const gate=initial.events.find((e:any)=>e.kind==='gate');
 assert.ok(before.outstanding.some(e=>e.key===gate.key));
 const changed={...source,events:initial.events.map((e:any)=>e.key===gate.key?{...e,title:e.title+' title edit',version:e.version+'-title-edit'}:e)};
 const after=await readCommanderConversation(async()=>page,id,changed);
 assert.ok(changed.events.some((e:any)=>e.key===gate.key));
 assert.ok(!after.outstanding.some(e=>e.key===gate.key));
});
for(const concurrentPublication of [false,true])test(`known 1647 item 7: response read discarded after concurrent publication=${concurrentPublication}`,async()=>{
 const listed=load('restart-after-list-timeline.json');
 const answered=load('restart-after-preview-timeline.json');
 const fold=await readCommanderConversation(async()=>listed,id,source);
 const result=fold.unresolved[0];
 const observer=new ChiefObserver(async()=>source);
 await observer.publish({} as any,id,result.batch,result.focus);
 let fetched!:()=>void,release!:(p:any)=>void;
 const waiting=new Promise<void>(r=>{fetched=r;});
 const page=new Promise<any>(r=>{release=r;});
 const api:any={agents:{ref:()=>({timeline:{refetch:async()=>{fetched();return page;}}})}};
 const receiving=observer.receive(api,id);
 await waiting;
 if(concurrentPublication)await observer.publish({} as any,id,result.batch,result.focus);
 release(answered);await receiving;
 assert.equal(observer.state.phase,concurrentPublication?'calling':'quiet');
 const confirmed=await readCommanderConversation(async()=>answered,id,source);
 assert.ok(confirmed.responses.has(result.batch.id));
});
