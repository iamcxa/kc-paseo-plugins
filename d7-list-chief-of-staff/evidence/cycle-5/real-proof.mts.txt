import fs from "node:fs";
import assert from "node:assert/strict";
import { DaemonClient } from "../@getpaseo/client/dist/daemon-client.js";

 await client.invokePluginRpc("paseo-spacedock-bridge","gates.read",{workflowDirs:[D.root+"/fictional"],fresh:true});
 const listing=await api.agents.list({subscribe:{}});listing.subscription.subscribe({snapshot:({entries}:any)=>{for(const {agent}of entries)void watch(agent.id);},update:(m:any)=>{if(m.type==="agent_update"&&m.payload.kind==="upsert")void watch(m.payload.agent.id);}});subs.push(()=>listing.subscription.release());

 for(let n=0;n<110;n++){await new Promise(r=>setTimeout(r,3000));record.last=await client.invokePluginRpc("paseo-spacedock-bridge","chief.read",{});save();if(["calling","failed","unknown"].includes(record.last.phase))break;}

 console.log(JSON.stringify({phase:record.last.phase,previewStatus:chief.status,previewError:chief.lastError}));
}catch(e){record.error=String(e);save();console.error(e);process.exitCode=1;}
